text = input("Enter Text: ")
key = int(input("Enter Key: "))

# Encryption
cipher = ""
for ch in text:
    if ch.isalpha():
        s = 65 if ch.isupper() else 97
        cipher += chr((ord(ch)-s+key)%26+s)
    else:
        cipher += ch

print("\nEncrypted:", cipher)

# Brute Force
print("\nBrute Force:")
for k in range(26):
    plain = ""
    for ch in cipher:
        if ch.isalpha():
            s = 65 if ch.isupper() else 97
            plain += chr((ord(ch)-s-k)%26+s)
        else:
            plain += ch
    print("Key", k, ":", plain)

# Decryption
